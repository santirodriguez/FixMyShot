cmake_minimum_required(VERSION 3.22.1)
project(fixmyshot_avif_reference LANGUAGES C)

if(NOT ANDROID)
    message(FATAL_ERROR "This source-build target is for the Android NDK only")
endif()
if(NOT DEFINED LIBAVIF_SOURCE_ROOT OR NOT EXISTS "${LIBAVIF_SOURCE_ROOT}/include/avif/avif.h")
    message(FATAL_ERROR "Set LIBAVIF_SOURCE_ROOT to a verified pinned checkout")
endif()
if(NOT EXISTS "${LIBAVIF_SOURCE_ROOT}/ext/aom/CMakeLists.txt")
    message(FATAL_ERROR "Missing pinned libaom checkout")
endif()

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(FIXMYSHOT_AVIF_DIRECT_DECODE "Enable synthetic-only direct 4:4:4 decode" OFF)
if(FIXMYSHOT_AVIF_DIRECT_DECODE AND NOT FIXMYSHOT_AVIF_PHYSICAL_PROBE)
    message(FATAL_ERROR "Direct decode belongs only to the isolated physical probe")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "Static dependency archives" FORCE)
set(AVIF_CODEC_AOM LOCAL CACHE STRING "Bundled pinned encoder" FORCE)
set(AVIF_CODEC_AOM_DECODE ${FIXMYSHOT_AVIF_DIRECT_DECODE} CACHE BOOL "Opt-in direct decoder" FORCE)
set(AVIF_CODEC_AOM_ENCODE ON CACHE BOOL "AV1 encoder" FORCE)
set(AVIF_LIBYUV OFF CACHE STRING "Avoid extra transitive packages" FORCE)
set(AVIF_LIBSHARPYUV OFF CACHE STRING "Unused" FORCE)
set(AVIF_JPEG OFF CACHE STRING "Unused" FORCE)
set(AVIF_ZLIBPNG OFF CACHE STRING "Unused" FORCE)
set(AVIF_BUILD_APPS OFF CACHE BOOL "No CLI from upstream" FORCE)
set(AVIF_BUILD_TESTS OFF CACHE BOOL "Probe uses its own fixture" FORCE)
set(AVIF_BUILD_MAN_PAGES OFF CACHE BOOL "No generated docs" FORCE)
set(AOM_TARGET_CPU generic CACHE STRING "Portable C baseline for feasibility" FORCE)
set(ENABLE_DOCS 0 CACHE INTERNAL "No aom docs")
set(ENABLE_EXAMPLES 0 CACHE INTERNAL "No aom examples")
set(ENABLE_TESTDATA 0 CACHE INTERNAL "No aom test data")
set(ENABLE_TESTS 0 CACHE INTERNAL "No aom tests")
set(ENABLE_TOOLS 0 CACHE INTERNAL "No aom tools")
set(FETCHCONTENT_SOURCE_DIR_LIBAOM "${LIBAVIF_SOURCE_ROOT}/ext/aom" CACHE PATH "Exact pinned libaom source" FORCE)
add_subdirectory("${LIBAVIF_SOURCE_ROOT}" "${CMAKE_BINARY_DIR}/libavif" EXCLUDE_FROM_ALL)

add_executable(avif_spike_cli avif_spike.c)
target_compile_options(avif_spike_cli PRIVATE -Wall -Wextra -Werror)
target_link_libraries(avif_spike_cli PRIVATE avif)
target_link_options(avif_spike_cli PRIVATE "-Wl,-z,max-page-size=16384")

# This JNI diagnostic is not linked to the production Android modules.
option(FIXMYSHOT_AVIF_JNI_PROBE "Build synthetic-only diagnostic JNI library" OFF)
if(FIXMYSHOT_AVIF_JNI_PROBE)
    add_library(avif_probe SHARED avif_probe_jni.c)
    target_compile_options(avif_probe PRIVATE -Wall -Wextra -Werror -fvisibility=hidden -ffunction-sections -fdata-sections)
    target_link_libraries(avif_probe PRIVATE avif)
    target_link_options(avif_probe PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,--gc-sections" "-Wl,--no-undefined")
endif()

# Separate opt-in physical-device diagnostic; never linked into production targets.
option(FIXMYSHOT_AVIF_PHYSICAL_PROBE "Build opt-in bitmap AVIF physical diagnostic" OFF)
if(FIXMYSHOT_AVIF_PHYSICAL_PROBE)
    add_library(avif_physical SHARED avif_physical_jni.c)
    if(FIXMYSHOT_AVIF_DIRECT_DECODE)
        target_compile_definitions(avif_physical PRIVATE FIXMYSHOT_AVIF_DIRECT_DECODE=1)
    endif()
    target_compile_options(avif_physical PRIVATE -Wall -Wextra -Werror -fvisibility=hidden)
    target_link_libraries(avif_physical PRIVATE avif jnigraphics)
    target_link_options(avif_physical PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,--no-undefined")
endif()
